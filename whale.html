import tkinter as tk # ボタンが押されたときの処理 def button_click(char): if char == 'C': entry.delete(0, tk.END) # クリア elif char == '=': try: # 入力された数式を計算 result = eval(entry.get()) entry.delete(0, tk.END) entry.insert(tk.END, str(result)) except Exception: entry.delete(0, tk.END) entry.insert(tk.END, "Error") # エラー処理 else: entry.insert(tk.END, char) # 数字や演算子を追加 # メインウィンドウの作成 root = tk.Tk() root.title("電卓") # ディスプレイ部分の作成 entry = tk.Entry(root, width=16, font=('Arial', 24), justify='right', bd=10) entry.grid(row=0, column=0, columnspan=4) # ボタンの配置設定 buttons = [ '7', '8', '9', '/', '4', '5', '6', '*', '1', '2', '3', '-', '0', 'C', '=', '+' ] # ボタンを画面に配置 row_val = 1 col_val = 0 for btn_text in buttons: action = lambda x=btn_text: button_click(x) tk.Button(root, text=btn_text, width=5, height=2, font=('Arial', 18), command=action).grid(row=row_val, column=col_val) col_val += 1 if col_val > 3: col_val = 0 row_val += 1 # アプリの起動 root.mainloop()
