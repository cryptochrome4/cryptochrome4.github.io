# LP Model Definition
import pulp

prob = pulp.LpProblem("LP_Model", pulp.LpMaximize)

x1 = pulp.LpVariable('x1', lowBound=0, cat='Continuous')
x2 = pulp.LpVariable('x2', lowBound=0, cat='Continuous')
x3 = pulp.LpVariable('x3', lowBound=0, cat='Continuous')
g = pulp.LpVariable('g', lowBound=0, cat='Continuous')
v = pulp.LpVariable('v', lowBound=0, cat='Continuous')

H = 1.0  
P_new = 250000
P_past = 260000

prob += (135000 + 20000 + 80000 + 6000 + 18000 + 6000 + 3000 + 21000 + 3000 + 3000 + 3000) * H + (350000 * 1.1) + 20000 + (1.15 * x1) + (1.20 * x2)

prob += x1 + x2 + x3 <= 110000
prob += x1 == 20000
prob += g + v == 10000
prob += g == 10000

status = prob.solve(pulp.PULP_CBC_CMD(msg=False))

print(f"x1 = {pulp.value(x1)}")
print(f"x2 = {pulp.value(x2)}")
print(f"x3 = {pulp.value(x3)}")
print(f"g = {pulp.value(g)}")
print(f"v = {pulp.value(v)}")

【Variables】
x1, x2, x3, g, v >= 0

【Constraints】
x1 + x2 + x3 <= 110000
x1 = 20000
P_new = 250000
P_past = 260000
P_new < P_past
g + v = 10000
g = 10000

【Objective Function】
Maximize R = 299000 * H + 385000 + 20000 + f(x1) + f(x2)

【Procedures】
1. 10000 + v = 10000 => v = 0
2. 20000 + x2 + x3 <= 110000 => x2 + x3 <= 90000
3. x3 = 0 => x2 = 90000 (under P_new = 250000)

【Optimal Solution】
x1 = 20000, x2 = 90000, x3 = 0, g = 10000, v = 0
